    <script>
        let logs = JSON.parse(localStorage.getItem('trip_time_logs')) || [];
        let expenses = JSON.parse(localStorage.getItem('trip_expense_logs')) || [];


        function switchTab(tabName) {
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
            
            if (tabName === 'time') {
                document.querySelectorAll('.tab-btn')[0].classList.add('active');
                document.getElementById('tab-time').classList.add('active');
            } else {
                document.querySelectorAll('.tab-btn')[1].classList.add('active');
                document.getElementById('tab-expense').classList.add('active');
            }
        }


        function forceRefresh() {
            if ('serviceWorker' in navigator) {
                navigator.serviceWorker.getRegistrations().then(registrations => {
                    for (let registration of registrations) registration.unregister();
                    window.location.reload(true);
                });
            } else {
                window.location.reload(true);
            }
        }


        function initTheme() {
            const savedTheme = localStorage.getItem('theme');
            if (savedTheme) {
                document.documentElement.setAttribute('data-theme', savedTheme);
                document.getElementById('theme-btn').textContent = savedTheme === 'dark' ? '☀️' : '🌙';
            } else if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
                document.documentElement.setAttribute('data-theme', 'dark');
                document.getElementById('theme-btn').textContent = '☀️';
            }
        }


        function toggleTheme() {
            const currentTheme = document.documentElement.getAttribute('data-theme');
            const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
            document.documentElement.setAttribute('data-theme', newTheme);
            localStorage.setItem('theme', newTheme);
            document.getElementById('theme-btn').textContent = newTheme === 'dark' ? '☀️' : '🌙';
        }


        initTheme();


        function getLocalDateString(d = new Date()) {
            const year = d.getFullYear();
            const month = String(d.getMonth() + 1).padStart(2, '0');
            const day = String(d.getDate()).padStart(2, '0');
            return `${year}-${month}-${day}`;
        }


        const sumDateInput = document.getElementById('summary-date');
        const expDateInput = document.getElementById('exp-date');
        if (sumDateInput) sumDateInput.value = getLocalDateString();
        if (expDateInput) expDateInput.value = getLocalDateString();


        function saveAndRender() {
            logs.sort((a, b) => new Date(`${b.date}T${b.time}`) - new Date(`${a.date}T${a.time}`));
            expenses.sort((a, b) => new Date(b.date) - new Date(a.date));
            
            localStorage.setItem('trip_time_logs', JSON.stringify(logs));
            localStorage.setItem('trip_expense_logs', JSON.stringify(expenses));
            
            renderTime();
            renderExpenses();
        }


        // TIME TRACKER FUNCTIONS
        function addTimestamp(status) {
            const now = new Date();
            logs.unshift({
                id: Date.now(),
                date: getLocalDateString(now),
                time: `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`,
                status: status,
                note: ''
            });
            saveAndRender();
        }


        function updateLog(id, field, value) {
            const log = logs.find(l => l.id === id);
            if (log) { log[field] = value; saveAndRender(); }
        }


        function deleteLog(id) {
            if (confirm("Delete this time entry?")) {
                logs = logs.filter(l => l.id !== id);
                saveAndRender();
            }
        }


        function getStatusColor(status) {
            switch(status) {
                case 'Active': return 'var(--active-color)';
                case 'Passive': return 'var(--passive-color)';
                case 'Break': return 'var(--break-color)';
                case 'End': return 'var(--end-color)';
                default: return '#64748b';
            }
        }


        function calculateDailySummary(selectedDate) {
            const sortedLogs = [...logs].sort((a, b) => new Date(`${a.date}T${a.time}`) - new Date(`${b.date}T${b.time}`));
            let totals = { Active: 0, Passive: 0, Break: 0 };


            for (let i = 0; i < sortedLogs.length - 1; i++) {
                const current = sortedLogs[i];
                const next = sortedLogs[i + 1];


                if (current.date === selectedDate) {
                    const diffMs = new Date(`${next.date}T${next.time}`) - new Date(`${current.date}T${current.time}`);
                    if (diffMs > 0) {
                        const hours = diffMs / (1000 * 60 * 60);
                        if (totals.hasOwnProperty(current.status) && current.status !== 'End') {
                            totals[current.status] += hours;
                        }
                    }
                }
            }
            return totals;
        }


        function renderTime() {
            const logList = document.getElementById('log-list');
            if (!logList) return;
            logList.innerHTML = logs.length === 0 ? '<div style="text-align:center; color:var(--subtext); font-size:0.85rem;">No timestamps logged.</div>' : '';


            logs.forEach(log => {
                const item = document.createElement('div');
                item.className = 'log-item';
                item.innerHTML = `
                    <div class="log-row-main">
                        <input type="date" value="${log.date}" onchange="updateLog(${log.id}, 'date', this.value)">
                        <input type="time" value="${log.time}" onchange="updateLog(${log.id}, 'time', this.value)">
                        <span class="badge" style="background:${getStatusColor(log.status)}">${log.status}</span>
                        <button class="delete-btn" onclick="deleteLog(${log.id})">✕</button>
                    </div>
                    <div class="log-row-note">
                        <input type="text" placeholder="Note (optional)" value="${log.note || ''}" onchange="updateLog(${log.id}, 'note', this.value)">
                    </div>
                `;
                logList.appendChild(item);
            });


            const selectedDate = document.getElementById('summary-date') ? document.getElementById('summary-date').value : getLocalDateString();
            const totals = calculateDailySummary(selectedDate);
            if (document.getElementById('sum-active')) document.getElementById('sum-active').textContent = totals.Active.toFixed(2) + ' h';
            if (document.getElementById('sum-passive')) document.getElementById('sum-passive').textContent = totals.Passive.toFixed(2) + ' h';
            if (document.getElementById('sum-break')) document.getElementById('sum-break').textContent = totals.Break.toFixed(2) + ' h';
            if (document.getElementById('sum-total')) document.getElementById('sum-total').textContent = (totals.Active + totals.Passive).toFixed(2) + ' h';
        }


        // EXPENSE TRACKER FUNCTIONS
        function addExpense() {
            const date = document.getElementById('exp-date').value;
            const amount = parseFloat(document.getElementById('exp-amount').value);
            const category = document.getElementById('exp-category').value;
            const payment = document.getElementById('exp-payment').value;
            const note = document.getElementById('exp-note').value;


            if (!date || isNaN(amount) || amount <= 0) {
                alert("Please enter a valid date and amount.");
                return;
            }


            expenses.unshift({
                id: Date.now(),
                date,
                amount,
                category,
                payment,
                note
            });


            document.getElementById('exp-amount').value = '';
            document.getElementById('exp-note').value = '';
            saveAndRender();
        }


        function updateExpense(id, field, value) {
            const exp = expenses.find(e => e.id === id);
            if (exp) {
                exp[field] = field === 'amount' ? parseFloat(value) || 0 : value;
                saveAndRender();
            }
        }


        function deleteExpense(id) {
            if (confirm("Delete this expense?")) {
                expenses = expenses.filter(e => e.id !== id);
                saveAndRender();
            }
        }


        function renderExpenses() {
            const list = document.getElementById('expense-list');
            if (!list) return;
            list.innerHTML = expenses.length === 0 ? '<div style="text-align:center; color:var(--subtext); font-size:0.85rem;">No expenses recorded.</div>' : '';


            let corpTotal = 0;
            let reimburseTotal = 0;


            expenses.forEach(exp => {
                if (exp.payment === 'Corporate Card') {
                    corpTotal += exp.amount;
                } else {
                    reimburseTotal += exp.amount;
                }


                const item = document.createElement('div');
                item.className = 'log-item';
                item.innerHTML = `
                    <div class="log-row-main">
                        <input type="date" value="${exp.date}" onchange="updateExpense(${exp.id}, 'date', this.value)">
                        <input type="number" step="0.01" style="width:75px;" value="${exp.amount.toFixed(2)}" onchange="updateExpense(${exp.id}, 'amount', this.value)">
                        <span class="badge badge-pay">${exp.payment === 'Corporate Card' ? 'Corp' : 'Self'}</span>
                        <button class="delete-btn" onclick="deleteExpense(${exp.id})">✕</button>
                    </div>
                    <div class="log-row-note">
                        <select onchange="updateExpense(${exp.id}, 'category', this.value)" style="flex:1;">
                            <option value="Meals" ${exp.category === 'Meals' ? 'selected' : ''}>Meals</option>
                            <option value="Transport" ${exp.category === 'Transport' ? 'selected' : ''}>Transport</option>
                            <option value="Hotel" ${exp.category === 'Hotel' ? 'selected' : ''}>Hotel</option>
                            <option value="Other" ${exp.category === 'Other' ? 'selected' : ''}>Other</option>
                        </select>
                        <select onchange="updateExpense(${exp.id}, 'payment', this.value)" style="flex:1;">
                            <option value="Corporate Card" ${exp.payment === 'Corporate Card' ? 'selected' : ''}>Corp Card</option>
                            <option value="Personal Card" ${exp.payment === 'Personal Card' ? 'selected' : ''}>Pers Card</option>
                            <option value="Cash" ${exp.payment === 'Cash' ? 'selected' : ''}>Cash</option>
                        </select>
                        <input type="text" placeholder="Note" value="${exp.note || ''}" onchange="updateExpense(${exp.id}, 'note', this.value)" style="flex:1.5;">
                    </div>
                `;
                list.appendChild(item);
            });


            if (document.getElementById('exp-sum-corp')) document.getElementById('exp-sum-corp').textContent = corpTotal.toFixed(2) + ' €';

            if (document.getElementById('exp-sum-total')) document.getElementById('exp-sum-total').textContent = (corpTotal + reimburseTotal).toFixed(2) + ' €';
        }


        // BACKUP & EXPORT
        function clearAllData() {
            if (confirm("Are you sure you want to clear ALL time logs and expenses?")) {
                logs = [];
                expenses = [];
                saveAndRender();
            }
        }


        function exportJsonBackup() {
            if (logs.length === 0 && expenses.length === 0) {
                alert("No data to export.");
                return;
            }
            const data = { logs, expenses };
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(data, null, 2));
            const anchor = document.createElement('a');
            anchor.setAttribute("href", dataStr);
            anchor.setAttribute("download", `tracker_backup_${getLocalDateString()}.json`);
            document.body.appendChild(anchor);
            anchor.click();
            anchor.remove();
        }


        function importJsonBackup(event) {
            const fileReader = new FileReader();
            fileReader.onload = function(e) {
                try {
                    const imported = JSON.parse(e.target.result);
                    if (confirm("Overwrite current data with this backup?")) {
                        if (Array.isArray(imported)) {
                            logs = imported;
                        } else {
                            logs = imported.logs || [];
                            expenses = imported.expenses || [];
                        }
                        saveAndRender();
                        alert("Backup restored!");
                    }
                } catch (err) {
                    alert("Error parsing JSON file.");
                }
            };
            if (event.target.files[0]) fileReader.readAsText(event.target.files[0]);
        }


        function exportNativeXlsx() {
            if (typeof XLSX === 'undefined') {
                alert("XLSX library not loaded. Make sure you have an active internet connection or local library.");
                return;
            }
            if (logs.length === 0 && expenses.length === 0) {
                alert("No data to export.");
                return;
            }


            const wb = XLSX.utils.book_new();


            if (logs.length > 0) {
                const grouped = {};
                logs.forEach(l => {
                    if (!grouped[l.date]) grouped[l.date] = [];
                    grouped[l.date].push(l);
                });
                const dates = Object.keys(grouped).sort();
                dates.forEach(d => grouped[d].sort((a, b) => a.time.localeCompare(b.time)));


                let maxRows = 0;
                dates.forEach(d => { if (grouped[d].length > maxRows) maxRows = grouped[d].length; });


                const sheetData = [dates];
                for (let i = 0; i < maxRows; i++) {
                    const row = [];
                    dates.forEach(d => {
                        const entry = grouped[d][i];
                        if (entry) {
                            const abbr = entry.status.charAt(0);
                            const noteStr = entry.note ? ` (${entry.note})` : '';
                            row.push(`${abbr} ${entry.time}${noteStr}`);
                        } else {
                            row.push('');
                        }
                    });
                    sheetData.push(row);
                }


                sheetData.push(dates.map(() => ''));
                const activeRow = [], passiveRow = [], breakRow = [], totalRow = [];
                dates.forEach(d => {
                    const totals = calculateDailySummary(d);
                    activeRow.push(`Active: ${totals.Active.toFixed(2)}h`);
                    passiveRow.push(`Passive: ${totals.Passive.toFixed(2)}h`);
                    breakRow.push(`Break: ${totals.Break.toFixed(2)}h`);
                    totalRow.push(`Total: ${(totals.Active + totals.Passive).toFixed(2)}h`);
                });
                sheetData.push(activeRow, passiveRow, breakRow, totalRow);


                const wsTime = XLSX.utils.aoa_to_sheet(sheetData);
                XLSX.utils.book_append_sheet(wb, wsTime, "Work Logs");
            }


            if (expenses.length > 0) {
                const expHeaders = [["Date", "Amount (€)", "Category", "Payment Method", "Note"]];
                const expRows = expenses.map(e => [e.date, e.amount, e.category, e.payment, e.note || '']);
                const wsExp = XLSX.utils.aoa_to_sheet(expHeaders.concat(expRows));
                XLSX.utils.book_append_sheet(wb, wsExp, "Expenses");
            }


            XLSX.writeFile(wb, `travel_report_${getLocalDateString()}.xlsx`);
        }
